node --experimental-strip-types -e 'import("./scripts/fixtures/listingSnapshotFixture.mjs").then((m) => { const e = m.writeFixtureSnapshot(process.argv[1]); console.log(Object.entries(e).map(([k, v]) => k + "=" + v).join(" ")); })' "$SNAP" > "$W/env.txt"
(cd apps/site && env $(cat "$W/env.txt") node ../../node_modules/astro/astro.js build --outDir "$OUT")
python3 -m http.server 4329 --bind 127.0.0.1 --directory "$OUT" >/dev/null 2>&1 & SERVER=$!
sleep 1; node scripts/fixtures/listingScreenshotCapture.mjs http://127.0.0.1:4329 "$SHOTS"; kill $SERVER
